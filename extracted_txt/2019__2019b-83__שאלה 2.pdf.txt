  define apply-select 
    (lambda (exps pred env index lst) 
      (if (null? exps) 
          lst 
          (let* ((e1 (value-of (call-exp pred (car exps)) env)) 
                 (newlst (list-val (append (expval->lst lst) (list index))))) 
            (if (expval->bool e1) 
                (apply-select (cdr exps) pred env (+ index 1) newlst) 
                (apply-select (cdr exps) pred env (+ index 1) lst))))))
