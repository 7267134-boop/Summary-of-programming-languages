;; Env ::= (empty-env) | (extend-env Var Schemeval Env)
;; empty-env : () -> Env

(lambda (env search-var)

((equal? (car env) ’empty-env)

((equal? (car env) ’extend-env)

(saved-val (caddr env))

saved-val
(apply-env saved-env search-var))))
(else
(report-invalid-env env)))))

תוכן מקוצר: נימוקים לחשיבות לימוד מבנה שפות תכנות וסקירה תהליכית ברמת-על של מנגנון פירוש התוכנית (Parsing & Interpreting).

הסבר: כאן נבנה המודל המנטלי הראשוני. תוכנית מחשב היא בתחילת דרכה לא יותר מרצף של טקסט (תווים). הפרק מסביר את תהליך ההמרה: הטקסט עובר דרך "מנתח קדמי" (Front-end) שהופך אותו ממחרוזת למבנה נתונים מסודר הנקרא "עץ תחביר מופשט" (AST). לאחר מכן, המבנה הזה עובר אל המפרש (Interpreter) אשר יוצק לתוכו משמעות (סמנטיקה) ומפיק תוצאה. הבנה זו קריטית להמשך, שכן לאורך הקורס ההתמקדות תהיה בכתיבת המפרש עצמו.

פרק 3: קבוצות נתונים אינדוקטיביות
עמ' 33 עד 44

תוכן מקוצר: תיאור והגדרה של קבוצות נתונים בצורה רקורסיבית (מלמטה-למעלה, מהטפחות-מטה, או באמצעות כללי היסק ודקדוק) ושימוש באינדוקציה מבנית לכתיבת קוד והוכחת טענות.

הסבר: עצים, רשימות ופסוקים הם מבנים המכילים בתוכם תת-מבנים מאותו סוג. כדי שהמפרש שלנו ידע לנתח עץ קוד, עלינו לדעת להגדיר בצורה מילולית ומתמטית איך העץ הזה נראה (באמצעות כתיבת BNF - דקדוק חסר הקשר). ברגע שמבינים שקוד המחשב הוא מבנה רקורסיבי (לדוגמה, ביטוי יכול להכיל בתוכו עוד ביטוי), קל להבין מדוע הפונקציות שאנו כותבים ב-Scheme כדי לעבד אותו (כמו פונקציה לבדיקת "משתנה חופשי" או ספירת אורך רשימה) חייבות להיות פונקציות רקורסיביות ש"משקפות" במדויק את אותו דקדוק.

פרק 4: הפשטת נתונים
עמ' 45 עד 90

תוכן מקוצר: יצירת חציצה בין "המפרט" (מה הנתון מייצג) ל"ייצוג" (איך הנתון נשמר מוחשית), שימוש בטיפוסי נתונים מופשטים כמו סביבות, ויישום דקדוק בתחביר מופשט במקום מוחשי.

הסבר: זהו פרק קריטי להנדסת תוכנה. הוא מסביר איך לבנות מערכות שבהן המפרש שלך "עיוור" למבנה הנתונים התחתון שבו שמרת את הזיכרון. במקום לגשת ישירות לרשימה או מערך, אתה מייצר ממשק התממשקות על ידי שימוש בבנאים (Constructors - למשל ליצירת סביבה ריקה או הרחבת סביבה), מנבאים (Predicates - שבודקים מאיזה סוג הנתון) ומחלצים (Extractors - שמוציאים מידע מהמבנה). הפרק מדגים זאת על מימושים שונים למספרים טבעיים (כגון קידוד Church) וסביבות (Environments), ומציג את הפקודות `define-datatype` ו-`cases` שהופכות כתיבה של עצי AST לקלה ומאורגנת בהרבה.

פרק 5: ביטויים
עמ' 91 עד 140

תוכן מקוצר: בנייה הדרגתית של סדרת מפרשים לשפות בעלות יכולות גדלות: שפת "ויהי" (הגדרת משתנים וחישובים פשוטים), "וישגור" (הוספת שגרות ופונקציות מסדר גבוה) ו"וייסוג" (הוספת תמיכה בשגרות רקורסיביות), תוך דיון בכריכה ותחימה (Scope/Binding).

הסבר: כאן אנו מתחילים לכתוב מפרשים שלמים. מתחילים בבסיס (חיבור, חיסור ומשתנים עם `let`), ולאחר מכן לומדים כיצד ליצור שגרות/פונקציות אמיתיות (`proc`). פה גם לומדים את המכניקה של סגירות (Closures) – הרעיון שכאשר פונקציה נוצרת, היא "אורזת" איתה בסביבה שלה את כל המשתנים שהיו קיימים באותו רגע בזיכרון, גם אם כשיקראו לה בעתיד הם כבר לא יהיו זמינים (תחימה לקסיקלית מול תחימה דינמית). לבסוף, לומדים טריקים מיוחדים כדי לאפשר לפונקציה לקרוא לעצמה מבלי לייצר "לולאה" שאינה נגמרת בזיכרון הסטטי (`letrec`).

פרק 6: המצב
עמ' 141 עד 166

תוכן מקוצר: הוספת תמיכה בפעולות בעלות השפעה (Side-effects/Mutation) באמצעות משתנים הניתנים לעדכון, דרך בניית אובייקט חסן (Store) המנהל כתובות זיכרון מפורשות ומרומזות.

הסבר: עד שלב זה, השפות שבנינו היו טהורות: אם `x` קיבל את הערך 5, הוא נשאר 5 לנצח. פרק זה מכניס את עולם המשתנים המצביים (כמו הפקודה `set!` או `x = x + 1`). כדי לעשות זאת, המפרש שלנו חייב כעת לייצר מבנה נתונים חדש הנקרא "החסן" – שפועל ממש כמו ה-RAM במחשב. נראה איך מעצבים שפה שבה המתכנת אחראי בעצמו ליצור כתובת ולהוציא ממנה נתונים (כמו פוינטרים בשפת C - הפניות מפורשות), ואיך מעצבים שפה חכמה יותר שבה כל משתנה הוא באופן אוטומטי מצביע אל הזיכרון בלי שהמתכנת יצטרך לציין זאת (הפניות מרומזות, כמו שעובדות Java ו-Python).

פרק 7: טיפוסים
עמ' 167 עד 200

תוכן מקוצר: הטמעת מערכות טיפוסים (Type Systems), ביצוע אנליזה ובדיקת טיפוסים סטטית במפרש (שפת "ויבדוק"), ואלגוריתם להסקה והקשת טיפוסים אוטומטית (שפת "ויקיש").

הסבר: במקום לחכות שהתוכנית תקרוס בזמן הריצה בגלל ניסיון לבצע חיבור בין מספר טבעי למחרוזת, הפרק הזה מלמד אותנו לבנות "בודק טיפוסים סטטי" שעובר על ה-AST לפני ההרצה עצמה (ממש כמו השלב של קומפילציה). לומדים איך בונים חוקי גזירה המאמתים שהקוד בטוח לריצה. בנוסף, הפרק עושה קפיצת מדרגה לעולם המודרני ומראה אלגוריתם שמאפשר לשפה "לנחש" ולהסיק לבד מהו סוג המשתנה, על ידי איסוף "אילוצים" ממשפטי הקוד ובניית מערכת משוואות (Unification Algorithm) שפותרת אותם מאחורי הקלעים כדי למצוא את הטיפוסים המדויקים, בלי שהמתכנת יצטרך לכתוב אותם במפורש.

פרק 8: פרדיגמות תכנות

פרק 12: המפרש של Scheme
עמ' 227 עד 238

תוכן מקוצר: עבודה מתקדמת מול מפרש השפה (ה-REPL) על ידי כתיבת פונקציות רקורסיביות, תנאים (cond) ופירוק רשימות.

הסבר: מכיוון שברמת המבנה הפנימי Scheme היא בעצם פשוט רשימה מקושרת ענקית, הפרק מלמד כיצד לבודד, לחתוך, ולשלוף תאים מתוך רשימות (הפונקציות המפורסמות `car`, `cdr` ו-`cons`). הוא מכין אותך לכתיבת קוד רקורסיבי מורכב שמטייל בתוך עמוקי המבנה של העץ, מה שיהווה את כלי העבודה העיקרי שלך בבניית המפרשים לקורס.
